# todo-scan Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Node/TypeScript CLI that lists `TODO:` markers grouped by file, with cyan `●` file links and yellow `▸` task links in a terminal, and that accepts either a directory or a single file.

**Architecture:** `match` extracts tasks from text. `format` renders the grouped string and owns color and OSC 8 links. `scan` reads one file or walks a directory. `cli` parses argv, classifies the path, chooses decoration, and writes stdout/stderr.

**Tech Stack:** Node 20+, TypeScript compiled with `tsc`, `node:test`, ESM, zero runtime dependencies.

## Global Constraints

- Package lives in `todo-scan/`. No runtime dependencies. TypeScript only compiles. Tests use `node:test`. Node 20 or newer. Package is ESM. `bin` maps `todo-scan` to `dist/cli.js`.
- Command is `todo-scan [caminho]`. With no argument, the root is `process.cwd()`.
- Decoration is on only when `process.stdout.isTTY === true` and `NO_COLOR` is absent or an empty string. Do not honor `FORCE_COLOR`.
- File line color is cyan: `\x1b[36m` … `\x1b[0m`. Task marker color is yellow: `\x1b[33m` … `\x1b[0m`.
- Hyperlink open is `\x1b]8;;<uri>\x1b\\`. Hyperlink close is `\x1b]8;;\x1b\\`. Color sits inside the hyperlink.
- URI example: `/Users/a b/src/app.ts` line 12 becomes `file:///Users/a%20b/src/app.ts:12`.
- Missing path stderr: `todo-scan: caminho não encontrado: <caminho>`.
- Not a file or directory stderr: `todo-scan: não é um arquivo nem um diretório: <caminho>`.
- Extra args stderr: `uso: todo-scan [caminho]`.
- Unreadable path stderr: `todo-scan: não foi possível ler <caminho>: <mensagem>`.
- Skip directory names exactly: `node_modules`, `.git`, `dist`, `build`, `coverage`. A null byte in the first 8192 bytes skips the whole file.
- Do not read `.gitignore`. Do not treat `FIXME`, `XXX`, or `HACK`. Do not print JSON, a count, or a report file. Finding tasks is exit 0.
- Every commit sets `GIT_AUTHOR_NAME=thiagonovato`, `GIT_AUTHOR_EMAIL=thiagonovato@gmail.com`, `GIT_COMMITTER_NAME=thiagonovato`, `GIT_COMMITTER_EMAIL=thiagonovato@gmail.com`. Do not change git config.
- Imports use the `.js` suffix required by `"module": "Node16"`.

export type Todo = { line: number; text: string };

export function todosInText(source: string): Todo[] {
  const todos: Todo[] = [];
  const lines = source.split(/\r?\n/);
  for (let i = 0; i < lines.length; i++) {
    const at = lines[i].indexOf("TODO:");
    if (at === -1) continue;
    todos.push({ line: i + 1, text: lines[i].slice(at + "TODO:".length).trim() });
  }
  return todos;
}
```

import type { Todo } from "./match.js";

export type FileTodos = { file: string; absolute: string; todos: Todo[] };

const ESC = "\u001b";

export function formatTodos(files: FileTodos[], decorate: boolean): string {
  const sorted = [...files].sort((a, b) => (a.file < b.file ? -1 : a.file > b.file ? 1 : 0));
  if (sorted.length === 0) return "";
  return sorted.map((file) => renderFile(file, decorate)).join("\n\n") + "\n";
}

function renderFile(file: FileTodos, decorate: boolean): string {
  const head = `● ${file.file}`;
  const tasks = file.todos.map((todo) => `  ▸ ${todo.line}  ${todo.text}`);
  if (!decorate) return [head, ...tasks].join("\n");
  const linkedHead = link(fileUri(file.absolute, 1), color(36, head));
  const linkedTasks = file.todos.map((todo) =>
    link(fileUri(file.absolute, todo.line), `  ${color(33, "▸")} ${todo.line}  ${todo.text}`),
  );
  return [linkedHead, ...linkedTasks].join("\n");
}

function color(code: number, text: string): string {
  return `${ESC}[${code}m${text}${ESC}[0m`;
}

function link(uri: string, visible: string): string {
  return `${ESC}]8;;${uri}${ESC}\\${visible}${ESC}]8;;${ESC}\\`;
}

function fileUri(absolute: string, line: number): string {
  const slashed = absolute.replaceAll("\\", "/");
  const encoded = slashed
    .split("/")
    .map((segment) => encodeURIComponent(segment).replace(/[!'()*]/g, hex))
    .join("/");
  return `file://${encoded}:${line}`;
}

function hex(char: string): string {
  return `%${char.charCodeAt(0).toString(16).toUpperCase()}`;
}
```

```ts
import { readdir, readFile } from "node:fs/promises";
import path from "node:path";
import type { FileTodos } from "./format.js";
import { todosInText } from "./match.js";

export type ScanIssue = { path: string; message: string };

const SKIP = new Set(["node_modules", ".git", "dist", "build", "coverage"]);

export class UnreadablePathError extends Error {
  readonly path: string;
  constructor(path: string, message: string) {
    super(message);
    this.name = "UnreadablePathError";
    this.path = path;
  }
}

export async function scanDirectory(root: string): Promise<{ files: FileTodos[]; issues: ScanIssue[] }> {
  const files: FileTodos[] = [];
  const issues: ScanIssue[] = [];
  await walk(root, root, files, issues, true);
  return { files, issues };
}

async function walk(
  dir: string,
  root: string,
  files: FileTodos[],
  issues: ScanIssue[],
  isRoot: boolean,
): Promise<void> {
  let entries;
  try {
    entries = await readdir(dir, { withFileTypes: true });
  } catch (error) {
    if (isRoot) throw new UnreadablePathError(dir, errorMessage(error));
    issues.push({ path: dir, message: errorMessage(error) });
    return;
  }
  for (const entry of entries) {
    if (entry.isSymbolicLink()) continue;
    const absolute = path.join(dir, entry.name);
    if (entry.isDirectory()) {
      if (SKIP.has(entry.name)) continue;
      await walk(absolute, root, files, issues, false);
      continue;
    }
    if (!entry.isFile()) continue;
    let buffer: Buffer;
    try {
      buffer = await readFile(absolute);
    } catch (error) {
      issues.push({ path: absolute, message: errorMessage(error) });
      continue;
    }
    if (buffer.subarray(0, 8192).includes(0)) continue;
    const todos = todosInText(buffer.toString("utf8"));
    if (todos.length === 0) continue;
    files.push({ file: path.relative(root, absolute), absolute, todos });
  }
}

function errorMessage(error: unknown): string {
  return error instanceof Error ? error.message : String(error);
}
```

import { chmod, mkdtemp, mkdir, writeFile } from "node:fs/promises";
import { tmpdir } from "node:os";
import path from "node:path";
import test from "node:test";
import { main } from "../src/cli.js";

function capture() {
  const out: string[] = [];
  const err: string[] = [];
  return {
    stdout: { write: (chunk: string) => void out.push(chunk), isTTY: false as boolean | undefined },
    stderr: { write: (chunk: string) => void err.push(chunk) },
    text: () => out.join(""),
    error: () => err.join(""),
  };
}

async function tempRoot(): Promise<string> {
  return mkdtemp(path.join(tmpdir(), "todo-scan-cli-"));
}

test("uses cwd when no path is given", async () => {
  const cwd = await tempRoot();
  await writeFile(path.join(cwd, "a.ts"), "TODO: daqui\n");
  const io = capture();
  const code = await main([], cwd, io.stdout, io.stderr, {});
  assert.equal(code, 0);
  assert.equal(io.text(), "● a.ts\n  ▸ 1  daqui\n");
  assert.equal(io.error(), "");
});

test("scans one file and reports missing, odd, and unreadable paths", async () => {
  const cwd = await tempRoot();
  const file = path.join(cwd, "only.ts");
  await writeFile(file, "TODO: só este\n");
  const io = capture();
  assert.equal(await main([file], cwd, io.stdout, io.stderr, {}), 0);
  assert.match(io.text(), /● .*only\.ts\n {2}▸ 1 {2}só este\n/);

  const missing = capture();
  assert.equal(await main(["nope.ts"], cwd, missing.stdout, missing.stderr, {}), 1);
  assert.equal(missing.error(), "todo-scan: caminho não encontrado: nope.ts\n");

  const fifo = path.join(cwd, "pipe");
  execFileSync("mkfifo", [fifo]);
  const odd = capture();
  assert.equal(await main([fifo], cwd, odd.stdout, odd.stderr, {}), 1);
  assert.equal(odd.error(), `todo-scan: não é um arquivo nem um diretório: ${fifo}\n`);

  const locked = path.join(cwd, "locked.ts");
  await writeFile(locked, "TODO: secreto\n");
  await chmod(locked, 0);
  const denied = capture();
  assert.equal(await main([locked], cwd, denied.stdout, denied.stderr, {}), 1);
  assert.match(denied.error(), new RegExp(`^todo-scan: não foi possível ler ${locked}: `));
  await chmod(locked, 0o644);
});

test("rejects extra arguments and warns for an unreadable nested file", async () => {
  const cwd = await tempRoot();
  const extra = capture();
  assert.equal(await main(["a", "b"], cwd, extra.stdout, extra.stderr, {}), 1);
  assert.equal(extra.error(), "uso: todo-scan [caminho]\n");

  await mkdir(path.join(cwd, "ok"));
  await writeFile(path.join(cwd, "ok", "seen.ts"), "TODO: visível\n");
  await mkdir(path.join(cwd, "secret"));
  await writeFile(path.join(cwd, "secret", "hidden.ts"), "TODO: escondido\n");
  await chmod(path.join(cwd, "secret"), 0);
  const io = capture();
  const code = await main([cwd], cwd, io.stdout, io.stderr, {});
  assert.equal(code, 0);
  assert.match(io.text(), /● ok\/seen\.ts/);
  assert.match(io.error(), /todo-scan: não foi possível ler /);
  await chmod(path.join(cwd, "secret"), 0o755);
});

test("decorates only a TTY without NO_COLOR", async () => {
  const cwd = await tempRoot();
  await writeFile(path.join(cwd, "a.ts"), "TODO: cor\n");
  const plain = capture();
  plain.stdout.isTTY = true;
  await main([], cwd, plain.stdout, plain.stderr, { NO_COLOR: "1" });
  assert.equal(plain.text().includes("\u001b"), false);

  const decorated = capture();
  decorated.stdout.isTTY = true;
  await main([], cwd, decorated.stdout, decorated.stderr, {});
  assert.match(decorated.text(), /\u001b\[36m● a\.ts\u001b\[0m/);
  assert.match(decorated.text(), /file:\/\/.*a\.ts:1/);
});

#!/usr/bin/env node
import { stat } from "node:fs/promises";
import path from "node:path";
import { pathToFileURL } from "node:url";
import { formatTodos, type FileTodos } from "./format.js";
import { scanDirectory, scanFile, UnreadablePathError } from "./scan.js";

type Writer = { write: (chunk: string) => void; isTTY?: boolean };

export async function main(
  argv: string[],
  cwd: string,
  stdout: Writer,
  stderr: Writer,
  env: NodeJS.ProcessEnv,
): Promise<number> {
  if (argv.length > 1) {
    stderr.write("uso: todo-scan [caminho]\n");
    return 1;
  }
  const raw = argv[0] ?? cwd;
  const resolved = path.resolve(cwd, raw);
  let info;
  try {
    info = await stat(resolved);
  } catch (error) {
    const message = error instanceof Error ? error.message : String(error);
    const missing = typeof error === "object" && error !== null && "code" in error && error.code === "ENOENT";
    stderr.write(
      missing
        ? `todo-scan: caminho não encontrado: ${raw}\n`
        : `todo-scan: não foi possível ler ${raw}: ${message}\n`,
    );
    return 1;
  }
  if (!info.isFile() && !info.isDirectory()) {
    stderr.write(`todo-scan: não é um arquivo nem um diretório: ${raw}\n`);
    return 1;
  }
  const decorate = stdout.isTTY === true && !env.NO_COLOR;
  try {
    const files = info.isDirectory()
      ? await filesInDirectory(resolved, stderr)
      : await scanFile(resolved, displayPath(cwd, resolved));
    stdout.write(formatTodos(files, decorate));
    return 0;
  } catch (error) {
    if (error instanceof UnreadablePathError) {
      stderr.write(`todo-scan: não foi possível ler ${raw}: ${error.message}\n`);
      return 1;
    }
    throw error;
  }
}

async function filesInDirectory(resolved: string, stderr: Writer): Promise<FileTodos[]> {
  const result = await scanDirectory(resolved);
  for (const issue of result.issues) {
    stderr.write(`todo-scan: não foi possível ler ${issue.path}: ${issue.message}\n`);
  }
  return result.files;
}

function displayPath(cwd: string, resolved: string): string {
  const relative = path.relative(cwd, resolved);
  if (relative.startsWith("..")) return resolved;
  return relative;
}

const entry = process.argv[1] ? pathToFileURL(process.argv[1]).href : "";
if (import.meta.url === entry) {
  const code = await main(process.argv.slice(2), process.cwd(), process.stdout, process.stderr, process.env);
  process.exit(code);
}
